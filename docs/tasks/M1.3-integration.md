**`podlator run <url>` 实现：**

```python
import asyncio
import uuid

@app.command()
def run(
    url: str = typer.Argument(..., help="YouTube 或播客 URL"),
    output_dir: str | None = typer.Option(None, help="输出目录"),
) -> None:
    """处理单个 URL，产出中文简报。"""
    asyncio.run(_run_pipeline(url, output_dir))


async def _run_pipeline(url: str, output_dir: str | None) -> None:
    """异步执行 pipeline。"""
    from podlator.config import Settings
    from podlator.graph.builder import build_graph
    from podlator.storage.db import TaskStore
    from podlator.logging import setup_logging

    settings = Settings()
    setup_logging(settings.log_level, settings.log_json_enabled)
    
    # 1. 创建任务记录
    store = TaskStore(settings.database_path)
    await store.initialize()
    
    task_id = str(uuid.uuid4())
    await store.create(task_id, url)
    
    typer.echo(f"任务创建: {task_id}")
    typer.echo(f"处理 URL: {url}")
    
    # 2. 构建并执行 Graph
    graph = build_graph()
    
    initial_state = {
        "task_id": task_id,
        "source_url": url,
        "status": "running",
        "current_node": "",
        "node_durations_ms": {},
        "total_cost_usd": 0.0,
    }
    
    try:
        # 执行 pipeline
        final_state = await graph.ainvoke(initial_state)
        
        # 3. 更新数据库
        await store.update(
            task_id,
            status="completed",
            title=final_state.get("title"),
            brief_path=final_state.get("output_path"),
            cost_usd=final_state.get("total_cost_usd", 0.0),
            duration_seconds=final_state.get("duration_seconds"),
        )
        
        output_path = final_state.get("output_path", "unknown")
        cost = final_state.get("total_cost_usd", 0.0)
        
        typer.echo(f"\n✅ 处理完成!")
        typer.echo(f"   简报: {output_path}")
        typer.echo(f"   费用: ${cost:.4f}")
        
    except Exception as e:
        await store.update(task_id, status="failed", error_message=str(e))
        typer.echo(f"\n❌ 处理失败: {e}", err=True)
        raise typer.Exit(code=1) from e
```

**`podlator status` 和 `podlator list` 实现：**

```python
@app.command()
def status(task_id: str | None = typer.Argument(None, help="任务 ID")) -> None:
    """查看任务状态。"""
    asyncio.run(_show_status(task_id))

async def _show_status(task_id: str | None) -> None:
    settings = Settings()
    store = TaskStore(settings.database_path)
    await store.initialize()
    
    if task_id:
        task = await store.get(task_id)
        if not task:
            typer.echo(f"任务不存在: {task_id}", err=True)
            raise typer.Exit(code=1)
        _print_task(task)
    else:
        tasks = await store.list_tasks(limit=1)
        if not tasks:
            typer.echo("暂无任务")
        else:
            _print_task(tasks[0])

def _print_task(task: dict) -> None:
    typer.echo(f"ID:     {task['id']}")
    typer.echo(f"标题:   {task.get('title', '(未知)')}")
    typer.echo(f"状态:   {task['status']}")
    typer.echo(f"节点:   {task.get('current_node', '-')}")
    typer.echo(f"费用:   ${task.get('cost_usd', 0):.4f}")
    typer.echo(f"创建:   {task['created_at']}")
    if task.get('error_message'):
        typer.echo(f"错误:   {task['error_message']}")
    if task.get('brief_path'):
        typer.echo(f"简报:   {task['brief_path']}")
```

## Phase 2: API 路由完善

### 2.1 更新 `src/podlator/api/routes.py`

将 501 占位的路由实现为真实逻辑：

**`POST /api/tasks/{task_id}/retry`**：
- 检查任务存在且状态为 `failed`
- 重置状态为 `pending`，清除 error_message
- 触发 graph 重新执行（后台任务）
- 返回更新后的 task

**`GET /api/tasks/{task_id}/brief`**：
- 检查任务存在且状态为 `completed`
- 读取 brief_path 指向的 Markdown 文件
- 返回 `TaskBriefResponse`

### 2.2 后台任务执行

`POST /api/tasks` 在 M0 只创建数据库记录。现在需要**同时触发 pipeline 执行**。

使用 FastAPI 的 `BackgroundTasks`：

```python
from fastapi import BackgroundTasks

@router.post("/api/tasks", status_code=201)
async def create_task(
    body: TaskCreate,
    background_tasks: BackgroundTasks,
    request: Request,
) -> TaskResponse:
    store = request.app.state.store
    task_id = str(uuid.uuid4())
    task = await store.create(task_id, str(body.url))
    
    # 在后台执行 pipeline
    background_tasks.add_task(run_pipeline_background, task_id, str(body.url), store)
    
    return TaskResponse(**task)


async def run_pipeline_background(task_id: str, url: str, store: TaskStore) -> None:
    """后台执行 pipeline。"""
    from podlator.graph.builder import build_graph
    
    await store.update(task_id, status="running")
    graph = build_graph()
    
    initial_state = {
        "task_id": task_id,
        "source_url": url,
        "status": "running",
        "current_node": "",
        "node_durations_ms": {},
        "total_cost_usd": 0.0,
    }
    
    try:
        final_state = await graph.ainvoke(initial_state)
        await store.update(
            task_id,
            status="completed",
            title=final_state.get("title"),
            brief_path=final_state.get("output_path"),
            cost_usd=final_state.get("total_cost_usd", 0.0),
            duration_seconds=final_state.get("duration_seconds"),
        )
    except Exception as e:
        logger.error("pipeline_failed", task_id=task_id, error=str(e), exc_info=True)
        await store.update(task_id, status="failed", error_message=str(e))
```
